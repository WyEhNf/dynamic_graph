cmake_minimum_required(VERSION 3.20)
project(aicore_streams_candidate64 LANGUAGES CXX)
set(PROJECT27_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../..")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(AICORE_SOLUTION_SOURCE "${PROJECT27_ROOT}/research/batch64/solution.cpp" CACHE FILEPATH "Single-file implementation used by local harness")
add_library(solution STATIC "${AICORE_SOLUTION_SOURCE}")
target_include_directories(solution PUBLIC "${PROJECT27_ROOT}/include")
add_library(simulator STATIC ${PROJECT27_ROOT}/src/simulator.cpp)
target_include_directories(simulator PUBLIC "${PROJECT27_ROOT}/include")
add_library(candidate_search STATIC ${PROJECT27_ROOT}/src/candidate_search.cpp)
target_include_directories(candidate_search PUBLIC "${PROJECT27_ROOT}/include")
target_link_libraries(candidate_search PUBLIC solution simulator)
foreach(t solution simulator)
  if(MSVC)
    target_compile_options(${t} PRIVATE /W4)
  else()
    target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
add_executable(aicore_demo ${PROJECT27_ROOT}/src/main.cpp)
target_link_libraries(aicore_demo PRIVATE candidate_search)
add_executable(aicore_bench ${PROJECT27_ROOT}/src/benchmark.cpp)
target_link_libraries(aicore_bench PRIVATE solution simulator)
add_executable(aicore_oracle ${PROJECT27_ROOT}/src/parallel_oracle.cpp)
target_link_libraries(aicore_oracle PRIVATE simulator)
add_executable(frontier_probe EXCLUDE_FROM_ALL ${PROJECT27_ROOT}/src/frontier_probe.cpp)
target_include_directories(frontier_probe PRIVATE "${PROJECT27_ROOT}/include")
target_link_libraries(frontier_probe PRIVATE simulator)
enable_testing()
add_executable(simulator_tests ${PROJECT27_ROOT}/research/batch64/tests/test_simulator.cpp)
target_link_libraries(simulator_tests PRIVATE candidate_search)
add_test(NAME scheduler_semantics COMMAND simulator_tests)
add_executable(model_tests ${PROJECT27_ROOT}/research/batch64/tests/test_models.cpp)
target_link_libraries(model_tests PRIVATE solution simulator)
add_test(NAME submission_model_and_json COMMAND model_tests)
add_executable(assignment_tests ${PROJECT27_ROOT}/research/batch64/tests/test_assignment.cpp)
target_include_directories(assignment_tests PRIVATE "${PROJECT27_ROOT}/include")
add_test(NAME assignment_grouping_equivalence COMMAND assignment_tests)
add_executable(serial_frontier_tests ${PROJECT27_ROOT}/research/batch64/tests/test_serial_frontier.cpp)
target_include_directories(serial_frontier_tests PRIVATE "${PROJECT27_ROOT}/include")
target_link_libraries(serial_frontier_tests PRIVATE simulator)
add_test(NAME serial_frontier_exact_equivalence COMMAND serial_frontier_tests)
file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/test-reports)
if(EXISTS ${PROJECT27_ROOT}/data/quick.json)
  add_test(NAME local_batch_quick COMMAND aicore_bench --input ${PROJECT27_ROOT}/data/quick.json --output ${CMAKE_CURRENT_BINARY_DIR}/test-reports/quick-results.json --failures ${CMAKE_CURRENT_BINARY_DIR}/test-reports/quick-failures.json)
endif()

# Arbitrary-precision Python gold, independent from both C++ state machines.
add_executable(numeric_boundary_tests ${PROJECT27_ROOT}/research/batch64/tests/test_numeric.cpp)
target_include_directories(numeric_boundary_tests PRIVATE "${PROJECT27_ROOT}/include")
target_link_libraries(numeric_boundary_tests PRIVATE simulator)
foreach(suite numeric-controls numeric-parallel numeric-parallel-valid scaled-serial-controls)
  add_test(NAME ${suite}_int64 COMMAND numeric_boundary_tests
    ${PROJECT27_ROOT}/data/${suite}-batch42.json
    ${CMAKE_CURRENT_BINARY_DIR}/test-reports/${suite}-numeric.json)
endforeach()
add_executable(numeric_priority_tests ${PROJECT27_ROOT}/research/batch64/tests/test_numeric_priority.cpp)
target_include_directories(numeric_priority_tests PRIVATE "${PROJECT27_ROOT}/include")
add_test(NAME numeric_priority_exact_gold COMMAND numeric_priority_tests ${PROJECT27_ROOT}/data/priority-gold-batch42.txt)
